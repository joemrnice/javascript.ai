Project 3 — Build your own Promise (guided walkthrough)

Implementing a spec-compliant Promise from scratch is one of the best ways to actually understand the microtask queue, .then chaining, and state transitions. Here's a working (simplified but correct-behaving) implementation.

MyPromise.js
class MyPromise {
  static PENDING = "pending";
  static FULFILLED = "fulfilled";
  static REJECTED = "rejected";

  #state = MyPromise.PENDING;
  #value;
  #callbacks = []; // queued {onFulfilled, onRejected} pairs, for handlers attached before settling

  constructor(executor) {
    const resolve = value => this.#settle(MyPromise.FULFILLED, value);
    const reject = reason => this.#settle(MyPromise.REJECTED, reason);
    try {
      executor(resolve, reject);
    } catch (err) {
      reject(err);
    }
  }

  #settle(state, value) {
    if (this.#state !== MyPromise.PENDING) return; // a promise can only settle ONCE
    this.#state = state;
    this.#value = value;
    queueMicrotask(() => this.#callbacks.forEach(cb => this.#run(cb)));
  }

  #run({ onFulfilled, onRejected, resolve, reject }) {
    try {
      if (this.#state === MyPromise.FULFILLED) {
        resolve(onFulfilled ? onFulfilled(this.#value) : this.#value);
      } else {
        if (onRejected) resolve(onRejected(this.#value));
        else reject(this.#value);
      }
    } catch (err) {
      reject(err);
    }
  }

  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const callback = { onFulfilled, onRejected, resolve, reject };
      if (this.#state === MyPromise.PENDING) {
        this.#callbacks.push(callback); // queue it — settle() will run it later
      } else {
        queueMicrotask(() => this.#run(callback)); // already settled — still async, per spec
      }
    });
  }

  catch(onRejected) { return this.then(null, onRejected); }
}

// Usage — behaves like a real Promise, including chaining and microtask timing
new MyPromise((resolve) => setTimeout(() => resolve(42), 100))
  .then(v => v * 2)
  .then(v => console.log(v)); // 84
The core insight

Every .then() call must itself return a new promise, so chains can keep going. And every callback — even on an already-settled promise — must run asynchronously (via queueMicrotask), which is why .then() callbacks never run before the rest of the current synchronous code, no matter how "instant" the promise already was.

Project 4 — A minimal virtual DOM (guided walkthrough)

The idea behind React's rendering model, stripped to its essence: describe UI as plain JS objects, diff two versions, and patch only what changed.

mini-vdom.js
// 1. Describe UI as plain objects — a "virtual" element
function h(tag, props = {}, ...children) {
  return { tag, props, children };
}

// 2. Turn a virtual element into a REAL DOM node (only used for the first render)
function createDOMNode(vnode) {
  if (typeof vnode === "string") return document.createTextNode(vnode);
  const el = document.createElement(vnode.tag);
  for (const [key, value] of Object.entries(vnode.props)) {
    if (key.startsWith("on")) el.addEventListener(key.slice(2).toLowerCase(), value);
    else el.setAttribute(key, value);
  }
  vnode.children.forEach(child => el.appendChild(createDOMNode(child)));
  return el;
}

// 3. Diff two virtual trees and patch only the real differences
function patch(parent, newVNode, oldVNode, index = 0) {
  const existing = parent.childNodes[index];

  if (!oldVNode) { parent.appendChild(createDOMNode(newVNode)); return; }
  if (!newVNode) { existing?.remove(); return; }
  if (typeof newVNode !== typeof oldVNode || newVNode.tag !== oldVNode?.tag) {
    parent.replaceChild(createDOMNode(newVNode), existing);
    return;
  }
  if (typeof newVNode === "string") {
    if (newVNode !== oldVNode) existing.textContent = newVNode;
    return;
  }
  // Same tag — patch attributes, then recurse into children
  const maxLen = Math.max(newVNode.children.length, oldVNode.children.length);
  for (let i = 0; i < maxLen; i++) {
    patch(existing, newVNode.children[i], oldVNode.children[i], i);
  }
}

// Usage: describe state as a tree, re-render by diffing against the previous tree
let oldTree = h("ul", {}, h("li", {}, "Item 1"));
const root = document.getElementById("app");
root.appendChild(createDOMNode(oldTree));

function update() {
  const newTree = h("ul", {}, h("li", {}, "Item 1"), h("li", {}, "Item 2 — added!"));
  patch(root, newTree, oldTree);
  oldTree = newTree; // the new tree becomes the baseline for next time
}
This is a teaching model, not a production one

Real virtual DOM implementations add key-based reconciliation (so re-ordered list items are moved, not recreated), batching, fiber-style interruptible rendering, and much more. The version above captures the core idea — diff, then patch — which is enough to make React's mental model click.
